import json
import asyncio
from typing import Dict, Any

def extract_json_data(tool_result: Any) -> Any:
    """Safely extracts dictionary or list payload from string, list, or MCP TextContent."""
    if isinstance(tool_result, str):
        try:
            tool_result = json.loads(tool_result)
        except Exception:
            return {}
            
    # If MCP returned a list like [TextContent(text='{...}')] or [{...}]
    if isinstance(tool_result, list) and len(tool_result) > 0:
        first_item = tool_result[0]
        if hasattr(first_item, 'text'):  # Handle LangChain / MCP TextContent objects
            return extract_json_data(first_item.text)
        elif isinstance(first_item, dict):
            return first_item
            
    if isinstance(tool_result, dict):
        return tool_result
        
    return {}

async def infrastructure_agent(state: AgentState) -> Dict[str, Any]:
    """
    AGENT 4: CMDB & Metrics Health Expert (Layer 02 Asset Focus)
    Queries active utilization performance stats across ALL detected hostnames.
    """
    logger.info("-----------Display State - Infrastructure Agent-----------")
    logger.info(state)
    session_id = state["session_id"]
    incident_number = state["incident_number"]
    app_id = state.get("app_id")
    opened_at = state.get("opened_at")

    # Safely handle either a single string, comma-separated string, or a list of hostnames
    raw_hostname = state.get("hostname") or state.get("all_hostnames") or []
    if isinstance(raw_hostname, str):
        # Handle comma-separated strings or single string
        hostnames = [h.strip() for h in raw_hostname.split(",") if h.strip()]
    elif isinstance(raw_hostname, list):
        hostnames = raw_hostname
    else:
        hostnames = []

    logger.info(f"Health Profile for hostnames {hostnames} at {opened_at}")

    # Fallback if no hostnames were identified
    if not hostnames:
        return {
            "asset_details": {},
            "technical_evidence": "No valid hostnames identified for telemetry metrics."
        }

    # 1. Create async tasks for ALL hostnames in parallel
    tasks = []
    for host in hostnames:
        cpu_task = tools_dict["fetch_asset_cpu_metrics"].ainvoke({
            "incident_number": incident_number,
            "asset_id": host
        })
        mem_task = tools_dict["fetch_asset_memory_metrics"].ainvoke({
            "incident_number": incident_number,
            "asset_id": host
        })
        disk_task = tools_dict["fetch_asset_disk_metrics"].ainvoke({
            "incident_number": incident_number,
            "asset_id": host
        })
        tasks.extend([cpu_task, mem_task, disk_task])

    # 2. Run all tasks across all hosts concurrently
    raw_results = await asyncio.gather(*tasks)

    # 3. Process results grouped by hostname
    health_profile = {}
    evidence_summaries = []

    for i, host in enumerate(hostnames):
        # Retrieve the 3 corresponding results for this host
        cpu_res = raw_results[i * 3]
        mem_res = raw_results[i * 3 + 1]
        disk_res = raw_results[i * 3 + 2]

        # Extract dictionary payloads
        cpu_dict = extract_json_data(cpu_res)
        mem_dict = extract_json_data(mem_res)
        disk_dict = extract_json_data(disk_res)

        # Extract historic average stats
        cpu_historic = cpu_dict.get("historic", {}) if isinstance(cpu_dict, dict) else {}
        mem_historic = mem_dict.get("historic", {}) if isinstance(mem_dict, dict) else {}
        disk_historic = disk_dict.get("historic", {}) if isinstance(disk_dict, dict) else {}

        cpu_avg = cpu_historic.get("avg", "N/A") if isinstance(cpu_historic, dict) else "N/A"
        mem_avg = mem_historic.get("avg", "N/A") if isinstance(mem_historic, dict) else "N/A"

        # Extract disk_avg (checking 'historic' key or root 'avg' key)
        disk_avg = (
            disk_historic.get("avg") 
            or (disk_dict.get("avg") if isinstance(disk_dict, dict) else None) 
            or "N/A"
        )

        # Populate host-level health profile
        health_profile[host] = {
            "cpu_utilization": cpu_dict,
            "memory_utilization": mem_dict,
            "disk_utilization": disk_dict
        }

        # Added Disk Avg to the technical evidence string
        evidence_summaries.append(f"[{host}] CPU Avg: {cpu_avg}%, Memory Avg: {mem_avg}%, Disk Avg: {disk_avg}%")

    return {
        "asset_details": health_profile,
        "technical_evidence": " | ".join(evidence_summaries)
    }

import json
import asyncio
from typing import Dict, Any, List

async def infrastructure_agent(state: AgentState) -> Dict[str, Any]:
    """
    AGENT 4: CMDB & Metrics Health Expert (Layer 02 Asset Focus)
    Uses LLM decision-making to selectively fetch CPU, Memory, or Disk metrics 
    based on incident details.
    """
    logger.info("-----------Display State - Infrastructure Agent-----------")
    logger.info(state)
    session_id = state["session_id"]
    incident_number = state["incident_number"]
    app_id = state.get("app_id")
    opened_at = state.get("opened_at")

    # Incident text context for decision prompt
    short_desc = state.get("short_description", "")
    description = state.get("description", "")

    # Safely extract hostnames
    raw_hostname = state.get("hostname") or state.get("all_hostnames") or []
    if isinstance(raw_hostname, str):
        hostnames = [h.strip() for h in raw_hostname.split(",") if h.strip()]
    elif isinstance(raw_hostname, list):
        hostnames = raw_hostname
    else:
        hostnames = []

    logger.info(f"Health Profile for hostnames {hostnames} at {opened_at}")

    if not hostnames:
        return {
            "asset_details": {},
            "technical_evidence": "No valid hostnames identified for telemetry metrics."
        }

    # ---------------------------------------------------------------------
    # 1. Prepare Prompt & Call LLM via openai_client
    # ---------------------------------------------------------------------
    metrics_prompt = f"""
    You are an IT infrastructure triage expert. Analyze the following incident details and decide 
    which telemetry metrics (cpu, memory, disk) are required to troubleshoot the issue.

    Short Description: {short_desc}
    Description: {description}

    Rules:
    - Include "cpu" for slowness, high utilization, compute load, or application hanging.
    - Include "memory" for RAM issues, heap memory, memory leaks, or out-of-memory errors.
    - Include "disk" for filesystem full, storage, partition space, or mount point issues.
    - If the issue is generic or unclear, include all three: ["cpu", "memory", "disk"].

    Output JSON format ONLY (no markdown formatting, no extra text):
    {{"required_metrics": ["cpu", "memory", "disk"]}}
    """

    try:
        # Prompting LLM using your openai_client structure
        response = openai_client.chat.completions.create(
            model="claude-opus-4-6-v1",
            messages=[
                {"role": "user", "content": metrics_prompt}
            ],
            temperature=0,  # Set temperature to 0 for deterministic decision-making
            max_tokens=200,
            stream=False
        )
        response_str = response.choices[0].message.content

        # Clean JSON markdown formatting if present
        clean_str = response_str.replace("```json", "").replace("```", "").strip()
        parsed_decision = json.loads(clean_str)
        required_metrics = parsed_decision.get("required_metrics", ["cpu", "memory", "disk"])
    except Exception as e:
        logger.warning(f"LLM decision step failed or returned invalid JSON. Defaulting to all metrics. Error: {e}")
        required_metrics = ["cpu", "memory", "disk"]

    logger.info(f"Selected metrics for {incident_number}: {required_metrics}")

    # ---------------------------------------------------------------------
    # 2. Build dynamically selected Tool Tasks
    # ---------------------------------------------------------------------
    tasks = []
    task_mapping = []  # Tracks (hostname, metric_type) for mapping results back

    for host in hostnames:
        if "cpu" in required_metrics:
            tasks.append(tools_dict["fetch_asset_cpu_metrics"].ainvoke({
                "incident_number": incident_number,
                "asset_id": host
            }))
            task_mapping.append((host, "cpu"))

        if "memory" in required_metrics:
            tasks.append(tools_dict["fetch_asset_memory_metrics"].ainvoke({
                "incident_number": incident_number,
                "asset_id": host
            }))
            task_mapping.append((host, "memory"))

        if "disk" in required_metrics:
            tasks.append(tools_dict["fetch_asset_disk_metrics"].ainvoke({
                "incident_number": incident_number,
                "asset_id": host
            }))
            task_mapping.append((host, "disk"))

    if not tasks:
        return {
            "asset_details": {},
            "technical_evidence": f"No telemetry metrics required based on incident context: '{short_desc}'"
        }

    # 3. Execute dynamically selected tasks concurrently
    raw_results = await asyncio.gather(*tasks)

    # ---------------------------------------------------------------------
    # 4. Process and group metric results by hostname
    # ---------------------------------------------------------------------
    health_profile = {host: {} for host in hostnames}

    for (host, metric_type), raw_res in zip(task_mapping, raw_results):
        parsed_data = extract_json_data(raw_res)
        health_profile[host][f"{metric_type}_utilization"] = parsed_data

    # Build technical evidence summaries dynamically
    evidence_summaries = []
    for host, metrics in health_profile.items():
        host_summary_parts = []
        
        if "cpu_utilization" in metrics:
            c_dict = metrics["cpu_utilization"]
            c_hist = c_dict.get("historic", {}) if isinstance(c_dict, dict) else {}
            c_avg = c_hist.get("avg", "N/A") if isinstance(c_hist, dict) else "N/A"
            host_summary_parts.append(f"CPU Avg: {c_avg}%")

        if "memory_utilization" in metrics:
            m_dict = metrics["memory_utilization"]
            m_hist = m_dict.get("historic", {}) if isinstance(m_dict, dict) else {}
            m_avg = m_hist.get("avg", "N/A") if isinstance(m_hist, dict) else "N/A"
            host_summary_parts.append(f"Memory Avg: {m_avg}%")

        if "disk_utilization" in metrics:
            d_dict = metrics["disk_utilization"]
            d_hist = d_dict.get("historic", {}) if isinstance(d_dict, dict) else {}
            d_avg = d_hist.get("avg") or (d_dict.get("avg") if isinstance(d_dict, dict) else None) or "N/A"
            host_summary_parts.append(f"Disk Avg: {d_avg}%")

        if host_summary_parts:
            evidence_summaries.append(f"[{host}] " + ", ".join(host_summary_parts))

    return {
        "asset_details": health_profile,
        "technical_evidence": " | ".join(evidence_summaries)
    }
