    lines.reverse()  # ES gave newest-first; flip to chronological order
    return "\n".join(lines) if lines else "No matching log events found in this window."


@require_GET
def evidence_metrics(request):
    """
    GET /api/triage/evidence/metrics/?incident_number=&asset_id=&metric=
        &range_start=&range_end=   (both optional — see below)

    range_start/range_end come from the frontend's time-range picker on the
    Live sub-tab (CPU/Memory/Disk/Network only — Raw Log never sends these).
    When both are present and parse successfully, they replace the default
    CURRENT_WINDOW trailing window for the "current" section. When absent
    or unparseable, behavior is unchanged from before this feature — the
    default trailing window is used, same as always.

    Response shape (matches the frontend's fetchEvidenceMetric() exactly):
        {
          "historic": { "avg": , "window_start": "", "window_end": "",
                         "top_processes": [{"process": "...", "cpu_percent": }, ...] },
          "current":  { "avg": , "captured_at": "",
                         "top_processes": [{"process": "...", "cpu_percent": }, ...] }
        }
    """
    incident_number = request.GET.get('incident_number')
    asset_id = request.GET.get('asset_id')
    metric = request.GET.get('metric')

    error_response = _validate_metrics_request(incident_number, asset_id, metric)
    if error_response:
        return error_response

    range_start, range_end = _parse_custom_range(request)

    if metric == 'cpu':
        return _build_cpu_metrics_response(incident_number, asset_id, range_start, range_end)
    if metric == 'memory':
        return _build_memory_metrics_response(incident_number, asset_id, range_start, range_end)
    if metric == 'disk':
        return _build_disk_metrics_response(incident_number, asset_id, range_start, range_end)
    if metric == 'network':
        return _build_network_metrics_response(incident_number, asset_id, range_start, range_end)
    if metric == 'raw':
        return _build_rawlog_metrics_response(incident_number, asset_id, range_start, range_end)

    # All five metrics (cpu/memory/disk/network/raw) are implemented above —
    # anything else is a genuinely bad request, not "not implemented yet".
    return JsonResponse({"error": f"metric='{metric}' is not a recognized metric"}, status=400)


def _parse_custom_range(request):
    """
    Parses optional range_start/range_end query params — ISO 8601 strings,
    as produced by the frontend's `new Date().toISOString()` — into
    timezone-aware datetimes. Returns (start, end), or (None, None) if
    either is missing or fails to parse. Callers treat (None, None) as
    "use the default window", so a malformed param degrades gracefully
    rather than erroring the whole request.
    """
    range_start_str = request.GET.get('range_start')
    range_end_str = request.GET.get('range_end')
    if not range_start_str or not range_end_str:
        return None, None
    start = parse_datetime(range_start_str)
    end = parse_datetime(range_end_str)
    if start is None or end is None:
        return None, None
    return start, end


def _validate_metrics_request(incident_number, asset_id, metric):
    """Returns a JsonResponse error if the request is invalid, else None."""
    if not incident_number or not asset_id or not metric:
        return JsonResponse(
            {"error": "incident_number, asset_id and metric are all required"},
            status=400,
        )
    return None


def _build_cpu_metrics_response(incident_number, asset_id, range_start=None, range_end=None):
    """Looks up the incident's open time, runs both ES queries, and builds the response."""
    try:
        incident_opened_at = get_incident_opened_at(incident_number)
    except NotImplementedError as e:
        logger.error(str(e))
        return JsonResponse({"error": "Server misconfiguration: incident lookup not implemented"}, status=500)

    if incident_opened_at is None:
        return JsonResponse({"error": f"Incident '{incident_number}' not found"}, status=404)

    historic = _get_historic_cpu_section(asset_id, incident_opened_at)
    if isinstance(historic, JsonResponse):
        return historic  # an error response bubbled up

    current = _get_current_cpu_section(asset_id, range_start, range_end)
    if isinstance(current, JsonResponse):
        return current  # an error response bubbled up

    return JsonResponse({"historic": historic, "current": current})


def _get_historic_cpu_section(asset_id, incident_opened_at):
    """Builds the 'historic' section: a window around when the incident opened."""
    hist_start = incident_opened_at - HISTORIC_WINDOW
    hist_end = incident_opened_at + HISTORIC_WINDOW
    try:
        hist_avg, hist_top = _query_top_processes_by_cpu(asset_id, hist_start, hist_end)
    except Exception:
        logger.exception("Elasticsearch query failed for historic CPU data (asset=%s)", asset_id)
        return JsonResponse({"error": "Failed to fetch historic CPU data from Elasticsearch"}, status=502)

    return {
        "avg": hist_avg,
        "window_start": hist_start.isoformat(),
        "window_end": hist_end.isoformat(),
        "top_processes": hist_top,
    }


def _get_current_cpu_section(asset_id, range_start=None, range_end=None):
    """
    Builds the 'current'/Live section: a custom time range from the
    frontend's picker if provided, otherwise the default trailing window.
    """
    if range_start and range_end:
        cur_start, now = range_start, range_end
    else:
        now = timezone.now()
        cur_start = now - CURRENT_WINDOW
    try:
        cur_avg, cur_top = _query_top_processes_by_cpu(asset_id, cur_start, now)
    except Exception:
        logger.exception("Elasticsearch query failed for current CPU data (asset=%s)", asset_id)
        return JsonResponse({"error": "Failed to fetch current CPU data from Elasticsearch"}, status=502)

    return {
        "avg": cur_avg,
        "captured_at": now.isoformat(),
        "top_processes": cur_top,
    }

    hist_start = incident_opened_at - HISTORIC_WINDOW
    hist_end = incident_opened_at + HISTORIC_WINDOW
    try:
        hist_avg, hist_top = _query_top_mount_points_by_disk_usage(asset_id, hist_start, hist_end)
    except Exception:
        logger.exception("Elasticsearch query failed for historic disk data (asset=%s)", asset_id)
        return JsonResponse({"error": "Failed to fetch historic disk data from Elasticsearch"}, status=502)

    return {
        "avg": hist_avg,
        "window_start": hist_start.isoformat(),
        "window_end": hist_end.isoformat(),
        "top_mount_points": hist_top,
    }

    }

    except NotImplementedError as e:

    if isinstance(current, JsonResponse):
        return current  # an error response bubbled up

    return JsonResponse({"historic": historic, "current": current})

def _snow_search_ci(asset_id):
    """
    Queries ServiceNow's Table API for the CMDB record matching asset_id.
    Uses sysparm_display_value=true so reference/choice fields (manufacturer,
    location, support_group, install_status, etc.) come back as
    human-readable text instead of raw sys_ids/choice codes.

    Returns the matching record dict, or None if nothing matched.
    """
    url = f"{SNOW_INSTANCE_URL}/api/now/table/{CMDB_TABLE}"
    params = {
        "sysparm_query": f"{ASSET_QUERY_FIELD}={asset_id}",
        "sysparm_limit": 1,
        "sysparm_display_value": "true",
    }
    response = requests.get(
        url,
        params=params,
        auth=SNOW_AUTH,
        # headers=SNOW_HEADERS,  # use this instead of `auth=` if you're on OAuth
        headers={"Accept": "application/json"},
        verify=SNOW_VERIFY_SSL,
        timeout=SNOW_REQUEST_TIMEOUT,
    )
    response.raise_for_status()
    results = response.json().get("result", [])
    return results[0] if results else None


def _extract_field(record, field_name):
    """
    ServiceNow fields come back as either a plain value or, for
    reference/choice fields with sysparm_display_value=true, an object
    like {"value": "...", "display_value": "..."}. Normalizes either into
    a plain value.
    """
    value = record.get(field_name)
    if isinstance(value, dict):
        return value.get("display_value") or value.get("value")
    return value


def _format_ram(raw_ram):
    """
    ServiceNow's `ram` field is typically a plain number in MB. Converts
    it to a human-readable string (e.g. "128 GB") for display.
    TODO: confirm your instance's actual unit — some customizations store
    this in GB already, in which case drop the /1024 conversion below.
    """
    if not raw_ram:
        return None
    try:
        mb = float(raw_ram)
    except (TypeError, ValueError):
        return str(raw_ram)  # already a string like "128 GB"? pass through as-is
    if mb >= 1024:
        return f"{mb / 1024:.0f} GB"
    return f"{mb:.0f} MB"


def _derive_retirement_status(record):
    """
    ServiceNow has no universal "retirement status" field — most instances
    encode this inside install_status's choice list (common OOB labels
    include "Retired", "Absent", "In Use", "In Maintenance", etc.).
    TODO: this is a keyword heuristic, not a guaranteed-correct mapping —
    adjust it (or point at a dedicated custom field, if your instance has
    one) to match your actual install_status choice list values.
    """
    install_status = (_extract_field(record, "install_status") or "").lower()
    if any(keyword in install_status for keyword in ("retire", "absent", "dispos")):
        return "Retired"
    return "Active"


def _build_asset_details_response(record):
    """Maps a raw ServiceNow CI record onto the frontend's expected field names."""
    data = {}
    for output_key, snow_field in SNOW_FIELD_MAP.items():
        data[output_key] = _extract_field(record, snow_field)

    data["ram"] = _format_ram(data.get("ram"))
    data["retirement_status"] = _derive_retirement_status(record)

    return data


def _is_windows_category(category_value):
    """
    OS-family check based on the CI's category field value.
    TODO: confirm what your `category` field actually holds — this does a
    case-insensitive "windows" substring check, adjust if your values look
    different (e.g. "Win2022", a class name, a custom choice list, etc.)
    """
    return "windows" in (category_value or "").lower()


def _snow_fetch_patch_date(asset_id, is_windows):
    """
    cmdb_ci_server doesn't carry patch data — it lives on the OS-specific
    extension table instead (cmdb_ci_win_server or cmdb_ci_linux_server,
    both of which extend cmdb_ci_server in ServiceNow's CMDB class
    hierarchy). Looks the same asset up there and returns just the patch
    date field.

    Returns the last_patch_date value, or None if no record/field found.
    Deliberately narrow (sysparm_fields) since this is a second round trip
    just to enrich one field.
    """
    table = CMDB_WIN_TABLE if is_windows else CMDB_LINUX_TABLE
    patch_field = WIN_LAST_PATCH_FIELD if is_windows else LINUX_LAST_PATCH_FIELD

    url = f"{SNOW_INSTANCE_URL}/api/now/table/{table}"
    params = {
        "sysparm_query": f"{ASSET_QUERY_FIELD}={asset_id}",
        "sysparm_limit": 1,
        "sysparm_display_value": "true",
        "sysparm_fields": f"{ASSET_QUERY_FIELD},{patch_field}",
    }
    response = requests.get(
        url,
        params=params,
        auth=SNOW_AUTH,
        headers={"Accept": "application/json"},
        verify=SNOW_VERIFY_SSL,
        timeout=SNOW_REQUEST_TIMEOUT,
    )
    response.raise_for_status()
    results = response.json().get("result", [])
    if not results:
        return None
    return _extract_field(results[0], patch_field)


@require_GET
def asset_details(request):
    """
    GET /api/triage/asset/details/?incident_number=&asset_id=

    incident_number isn't actually needed to look up the CMDB record (the
    asset is found purely by asset_id/hostname) — it's accepted and
    validated for consistency with the other evidence endpoints, and in
    case you want to log/audit which incident triggered the lookup.
    """
    incident_number = request.GET.get('incident_number')
    asset_id = request.GET.get('asset_id')

    if not incident_number or not asset_id:
        return JsonResponse({"error": "incident_number and asset_id are required"}, status=400)

    try:
        record = _snow_search_ci(asset_id)
    except Exception:
        logger.exception("ServiceNow lookup failed for asset=%s", asset_id)
        return JsonResponse({"error": "Failed to fetch asset details from ServiceNow"}, status=502)

    if record is None:
        return JsonResponse({"error": f"Asset '{asset_id}' not found in ServiceNow"}, status=404)

    data = _build_asset_details_response(record)

    # cmdb_ci_server has no patch data — fetch it from whichever OS-specific
    # extension table matches this CI's category. Deliberately non-fatal:
    # if this second lookup fails, the rest of the asset details still
    # render fine with last_patch_date left as null rather than the whole
    # request failing over one supplementary field.
    is_windows = _is_windows_category(_extract_field(record, CATEGORY_FIELD))
    try:
        patch_date = _snow_fetch_patch_date(asset_id, is_windows)
        if patch_date:
            data["last_patch_date"] = patch_date
    except Exception:
        logger.exception(
            "ServiceNow patch-date lookup failed for asset=%s (table=%s)",
            asset_id, CMDB_WIN_TABLE if is_windows else CMDB_LINUX_TABLE,
        )

    return JsonResponse(data)

CHANGE_REQUEST_TABLE = "change_request"
CHANGE_APP_FIELD = "u_application"   # TODO: field linking a change to an app/app_id — confirm your instance's real field

    "submitter": "requested_by",                   # TODO: or "sys_created_by"

    "change_coordinator": "u_change_coordinator",   # TODO: rarely OOB — or "assigned_to"

    "end_time": "end_date",
    "status": "state",                              # choice field -> display_value
}

# Holds the list of Service Requests raised for alarm suppression on this
# change — lives directly on change_request itself, no extra query needed.
ALARM_SUPPRESSION_FIELD = "u_alarm_suppression"

IMPACTED_CI_TABLE = "task_ci"                # TODO: confirm — ServiceNow's OOB task<->CI junction table
IMPACTED_CI_TASK_FIELD = "task"              # reference back to the change_request sys_id
IMPACTED_CI_FIELD = "ci_item"                # reference to the affected CI
SUPPRESSION_STATUS_FIELD = "u_suppression_status"            # TODO: almost certainly custom, may not exist
SUPPRESSION_REQUEST_FIELD = "u_suppression_request_number"   # TODO: almost certainly custom, may not exist


def _snow_search_changes(app_id, window_start, window_end):
    """
    Queries change_request for changes tied to app_id whose window
    overlaps [window_start, window_end]. "Overlaps" (not just "starts
    within") means: the change started before the window ends AND ended
    after the window starts — this also catches changes that began before
    the incident but were still running into the +/-24h window, not just
    ones that started inside it. Adjust to a simpler start_date-only check
    if you'd rather match your exact wording more literally.

    NOTE: window_start/window_end are Python datetimes (from Django's
    timezone-aware `now`/incident lookup) formatted as ServiceNow expects
    for GlideDateTime comparisons ("YYYY-MM-DD HH:MM:SS"). TODO: confirm
    your ServiceNow instance's configured timezone matches what these
    datetimes represent — if your instance stores times in a different
    timezone than Django's incident timestamps, these comparisons will be
    off by that offset.
    """
    url = f"{SNOW_INSTANCE_URL}/api/now/table/{CHANGE_REQUEST_TABLE}"
    query = (
        f"{CHANGE_APP_FIELD}={app_id}"
        f"^start_date<={window_end.strftime('%Y-%m-%d %H:%M:%S')}"
        f"^end_date>={window_start.strftime('%Y-%m-%d %H:%M:%S')}"
    )
    params = {
        "sysparm_query": query,
        "sysparm_display_value": "true",
        "sysparm_limit": 50,
    }
    response = requests.get(
        url,
        params=params,
        auth=SNOW_AUTH,
        headers={"Accept": "application/json"},
        verify=SNOW_VERIFY_SSL,
        timeout=SNOW_REQUEST_TIMEOUT,
    )
    response.raise_for_status()
    return response.json().get("result", [])


def _snow_search_impacted_cis(change_sys_id):
    """
    Looks up impacted CIs for one change via the task_ci junction table.
    Non-fatal by design in the caller below — a failed lookup here just
    means that one change shows an empty impacted-CI list, not a failed
    request overall.
    """
    url = f"{SNOW_INSTANCE_URL}/api/now/table/{IMPACTED_CI_TABLE}"
    params = {
        "sysparm_query": f"{IMPACTED_CI_TASK_FIELD}={change_sys_id}",
        "sysparm_display_value": "true",
        "sysparm_limit": 50,
    }
    response = requests.get(
        url,
        params=params,
        auth=SNOW_AUTH,
        headers={"Accept": "application/json"},
        verify=SNOW_VERIFY_SSL,
        timeout=SNOW_REQUEST_TIMEOUT,
    )
    response.raise_for_status()
    results = response.json().get("result", [])

    impacted = []
    for row in results:
        impacted.append({
            "ci_name": _extract_field(row, IMPACTED_CI_FIELD) or "Unknown",
            "suppression_status": _extract_field(row, SUPPRESSION_STATUS_FIELD) or "Not Suppressed",
            "suppression_request_number": _extract_field(row, SUPPRESSION_REQUEST_FIELD) or "N/A",
        })
    return impacted


def _parse_suppression_list(raw_value):
    """
    ServiceNow's u_alarm_suppression field holds the Service Requests
    raised for alarm suppression on this change. If it's a glide_list
    field, fetching with sysparm_display_value=true typically returns a
    single comma-separated display string (e.g. "REQ0012345, REQ0012346")
    rather than a JSON array — this splits that into a clean list.
    TODO: if your field is a plain text field with a different separator,
    or already comes back as a real list, adjust this parsing.
    """
    if not raw_value:
        return []
    if isinstance(raw_value, list):
        return [str(v).strip() for v in raw_value if str(v).strip()]
    return [part.strip() for part in str(raw_value).split(',') if part.strip()]


def _build_change_response(record, app_id):
    """Maps a raw ServiceNow change_request record onto the frontend's expected field names."""
    data = {
        output_key: _extract_field(record, snow_field)
        for output_key, snow_field in CHANGE_FIELD_MAP.items()
    }
    data["app_id"] = app_id
    data["alarm_suppression_requests"] = _parse_suppression_list(
        _extract_field(record, ALARM_SUPPRESSION_FIELD)
    )

    change_sys_id = record.get("sys_id")  # sys_id is never a reference field, always a plain string
    try:
        data["impacted_cis"] = _snow_search_impacted_cis(change_sys_id) if change_sys_id else []
    except Exception:
        logger.exception("Impacted-CI lookup failed for change=%s", data.get("change_number"))
        data["impacted_cis"] = []

    return data


@require_GET
def changes(request):
    """
    GET /api/triage/changes/?incident_number=&app_id=

    Fetches ServiceNow change requests tied to app_id whose window overlaps
    the +/-24h period around when the incident opened.

    Response: { "changes": [ {change_number, submitter, affected_ci,
    change_coordinator, start_time, end_time, status, app_id,
    impacted_cis: [...]}, ... ] }
    """
    incident_number = request.GET.get('incident_number')
    app_id = request.GET.get('app_id')

    if not incident_number or not app_id:
        return JsonResponse({"error": "incident_number and app_id are required"}, status=400)

    try:
        incident_opened_at = get_incident_opened_at(incident_number)
    except NotImplementedError as e:
        logger.error(str(e))
        return JsonResponse({"error": "Server misconfiguration: incident lookup not implemented"}, status=500)

    if incident_opened_at is None:
        return JsonResponse({"error": f"Incident '{incident_number}' not found"}, status=404)

    window_start = incident_opened_at - timedelta(hours=CHANGE_WINDOW_HOURS)
    window_end = incident_opened_at + timedelta(hours=CHANGE_WINDOW_HOURS)

    try:
        records = _snow_search_changes(app_id, window_start, window_end)
    except Exception:
        logger.exception("ServiceNow change lookup failed for app_id=%s", app_id)
        return JsonResponse({"error": "Failed to fetch changes from ServiceNow"}, status=502)

    changes_list = [_build_change_response(record, app_id) for record in records]

    return JsonResponse({"changes": changes_list})
