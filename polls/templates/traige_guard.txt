"""
triage_guard.py - protects the downstream triage service (FastAPI/uvicorn, few workers) from request bursts,
and guarantees that ONE incident is only ever triaged by ONE run at a time, whoever asks.

Drop this file into your Django app and put @triage_guard on the view that FORWARDS the "start triage" call:

    from .triage_guard import triage_guard, note_session_status

    @login_required
    @never_cache
    @csrf_exempt
    @triage_guard                                 # <- innermost: anonymous users are redirected first, 429s are never cached
    def trigger_triage(request):
        ...                                       # your existing code (calls FastAPI, returns JsonResponse(session_row))

    def session_status(request):
        ...                                       # your existing code
        note_session_status(session_id, triage_status)    # frees the incident when its run finishes

What it enforces, BEFORE anything is forwarded to FastAPI (all keyed in Django's cache, so it holds across users,
browsers and tabs):

  1. INCIDENT LOCK. While a run for incident X is in progress, every other request for X - any user, any browser -
     is NOT forwarded. It is handed the SAME run (200, header "X-Triage-Deduplicated: true"), so its page simply
     polls that run and shows the recommendation the one real run stores. If the first request is still being
     forwarded, duplicates wait (up to TRIAGE_GUARD_WAIT_SECONDS) for it and then get its run.
  2. RATE LIMIT per user/IP (burst protection)            -> 429 + Retry-After
  3. GLOBAL CONCURRENCY CAP across different incidents   -> 429 + Retry-After
A lock is held until the run finishes (note_session_status / release_for_session / the optional status checker /
a start call that already returns a finished run) or TRIAGE_GUARD_ACTIVE_TTL seconds pass, so a crashed run can
never block an incident forever.

IMPORTANT: the locks live in Django's cache and only work across processes/servers if CACHES['default'] (or the
alias in TRIAGE_GUARD_CACHE_ALIAS) is SHARED: Redis, Memcached, the database or file cache. With the default
LocMemCache every worker has its own copy and nothing is enforced across them.

Settings (all optional):
    TRIAGE_GUARD_MAX_ACTIVE            = 1     concurrent running triages (different incidents), system-wide
    TRIAGE_GUARD_ACTIVE_TTL            = 900   seconds before an unreleased lock frees itself
    TRIAGE_GUARD_WAIT_SECONDS          = 10    how long a duplicate waits for the first request to finish being forwarded
    TRIAGE_GUARD_STATUS_CHECKER        = None  "app.module.func": callable(session_id) -> status str or None. If set, a
                                               duplicate first verifies the run is really still running, so a lock whose
                                               release hook was missed can never serve a stale, finished run.
    TRIAGE_GUARD_REUSE_FINISHED_SECONDS= 0     >0: for this long after a run COMPLETES, a new trigger for the incident
                                               gets that finished run back instead of starting another (0 = always fresh)
    TRIAGE_GUARD_RATE_LIMIT            = 6     triggers per client per window
    TRIAGE_GUARD_RATE_WINDOW           = 60    seconds
    TRIAGE_GUARD_RETRY_AFTER           = 15    seconds advertised when the system is busy
    TRIAGE_GUARD_CACHE_ALIAS           = 'default'
    TRIAGE_GUARD_INCIDENT_PARAM        = 'incident_number'
"""
from __future__ import annotations

import functools
import json
import logging
import time
from typing import Any, Callable, Optional

from django.conf import settings
from django.core.cache import caches
from django.http import HttpRequest, HttpResponse, JsonResponse
from django.utils.module_loading import import_string

log = logging.getLogger(__name__)

_now: Callable[[], float] = time.time           # patched in tests
_sleep: Callable[[float], None] = time.sleep    # patched in tests
_monotonic: Callable[[], float] = time.monotonic
_PREFIX = "triage_guard:"
SUCCESS_STATUSES = {"completed", "complete", "success", "successful"}
FINISHED_STATUSES = SUCCESS_STATUSES | {"failed", "error"}


def _cfg(name: str, default: Any) -> Any:
    return getattr(settings, f"TRIAGE_GUARD_{name}", default)


def _cache():
    return caches[_cfg("CACHE_ALIAS", "default")]


def _incident_key(number: str) -> str:
    return f"{_PREFIX}incident:{number.upper()}"


def _done_key(number: str) -> str:
    return f"{_PREFIX}done:{number.upper()}"


def _slot_key(i: int) -> str:
    return f"{_PREFIX}slot:{i}"


def _session_key(session_id: Any) -> str:
    # GUIDs come back in different casings from different services: compare them case-insensitively
    return f"{_PREFIX}session:{str(session_id).strip().lower()}"


# --------------------------------------------------------------------------- responses
def _busy(message: str, retry_after: int, reason: str) -> JsonResponse:
    retry_after = max(1, int(retry_after))
    resp = JsonResponse({"status": "busy", "reason": reason, "error": message, "retry_after": retry_after}, status=429)
    resp["Retry-After"] = str(retry_after)
    return resp


def _same_run(payload: Any, why: str) -> JsonResponse:
    resp = JsonResponse(payload, safe=False, status=200)
    resp["X-Triage-Deduplicated"] = "true"
    resp["X-Triage-Dedup-Reason"] = why             # "running" (attached to the live run) | "finished" (recent completed run)
    return resp


# --------------------------------------------------------------------------- helpers
def _incident_of(request: HttpRequest) -> str:
    """The incident number from the query string, a form post, or a JSON body.

    The JSON body is parsed whatever the Content-Type says (like the triage view itself does:
    json.loads(request.body)), so a client that posts JSON as text/plain or without a header is still recognised.
    request.body is cached by Django, so the view can read it again afterwards.
    """
    name = _cfg("INCIDENT_PARAM", "incident_number")
    found = request.GET.get(name)
    if not found:
        try:
            raw = request.body
        except Exception:                                # e.g. RequestDataTooBig: let the view deal with it
            raw = b""
        if raw:
            try:
                body = json.loads(raw.decode(request.encoding or "utf-8"))
                found = body.get(name) if isinstance(body, dict) else None
            except Exception:
                found = None
            if not found and "json" not in (request.content_type or ""):
                found = request.POST.get(name)           # form-encoded post
    return str(found or "").strip()


def _client_of(request: HttpRequest) -> str:
    user = getattr(request, "user", None)
    if user is not None and getattr(user, "is_authenticated", False):
        return f"user:{user.pk}"
    # NOTE: behind a proxy make sure REMOTE_ADDR is the real client (trusted-proxy setup).
    return f"ip:{request.META.get('REMOTE_ADDR', 'unknown')}"


def _rate_limited(client: str) -> Optional[int]:
    """Seconds to wait if `client` exceeded the limit, else None. Fixed-window counter."""
    limit, window = int(_cfg("RATE_LIMIT", 6)), int(_cfg("RATE_WINDOW", 60))
    if limit <= 0:
        return None
    cache = _cache()
    now = _now()
    key = f"{_PREFIX}rate:{client}:{int(now // window)}"
    cache.add(key, 0, window + 1)
    try:
        count = cache.incr(key)
    except ValueError:                                   # the key expired between add and incr
        cache.add(key, 1, window + 1)
        count = 1
    return int(window - (now % window)) + 1 if count > limit else None


def _acquire_slot(number: str) -> Optional[int]:
    cache, ttl = _cache(), int(_cfg("ACTIVE_TTL", 900))
    for i in range(int(_cfg("MAX_ACTIVE", 1))):
        if cache.add(_slot_key(i), {"incident": number.upper(), "ts": _now()}, ttl):    # atomic: exactly one caller wins
            return i
    return None


def _is_finished(payload: Any) -> bool:
    if isinstance(payload, dict):
        for key in ("status", "triage_status", "name_trg_ssin_sts_type"):
            v = payload.get(key)
            if isinstance(v, str) and v.strip().lower() in FINISHED_STATUSES:
                return True
    return False


def _is_success(payload: Any) -> bool:
    if isinstance(payload, dict):
        for key in ("status", "triage_status", "name_trg_ssin_sts_type"):
            v = payload.get(key)
            if isinstance(v, str) and v.strip().lower() in SUCCESS_STATUSES:
                return True
    return False


def _remember_finished(number: str, payload: Any) -> None:
    reuse = int(_cfg("REUSE_FINISHED_SECONDS", 0))
    if reuse > 0 and payload is not None:
        _cache().set(_done_key(number), payload, reuse)


def _release(number: str, finished_ok: bool = False) -> None:
    """Free the incident lock and its slot. `finished_ok`: the run completed, so remember it if reuse is on."""
    cache = _cache()
    state = cache.get(_incident_key(number))
    if isinstance(state, dict):
        if state.get("slot") is not None:
            cache.delete(_slot_key(state["slot"]))
        if state.get("session_id"):
            cache.delete(_session_key(state["session_id"]))
        if finished_ok:
            _remember_finished(number, state.get("payload"))
    cache.delete(_incident_key(number))


def _release_claim(number: str, slot: int) -> None:
    cache = _cache()
    cache.delete(_slot_key(slot))
    cache.delete(_incident_key(number))


def _verified_running(state: dict) -> bool:
    """With a status checker configured, confirm the run is really still going (a missed release hook must never serve a stale run)."""
    path, session_id = _cfg("STATUS_CHECKER", None), state.get("session_id")
    if not path or not session_id:
        return True
    try:
        status = import_string(path)(session_id)
    except Exception:
        log.exception("triage_guard: status checker failed; assuming the run is still active")
        return True
    return not (isinstance(status, str) and status.strip().lower() in FINISHED_STATUSES)


# --------------------------------------------------------------------------- public API
def release_for_incident(number: str) -> None:
    """Free the lock held by this incident (idempotent)."""
    _release(number)


def release_for_session(session_id: str, finished_ok: bool = False) -> None:
    """Free the lock of the run with this session id (idempotent)."""
    number = _cache().get(_session_key(session_id))
    if number:
        _release(number, finished_ok=finished_ok)


def note_session_status(session_id: str, triage_status: str) -> bool:
    """Call from the session-status view. Frees the incident once its run has finished; returns True if it did."""
    status = (triage_status or "").strip().lower()
    if status in FINISHED_STATUSES:
        release_for_session(session_id, finished_ok=status in SUCCESS_STATUSES)
        return True
    return False


def triage_guard(view: Callable[..., HttpResponse]) -> Callable[..., HttpResponse]:
    @functools.wraps(view)
    def wrapper(request: HttpRequest, *args: Any, **kwargs: Any) -> HttpResponse:
        if request.method != "POST":
            return view(request, *args, **kwargs)
        number = _incident_of(request)
        log.debug("triage_guard: incident=%r method=%s", number, request.method)
        if not number:
            return view(request, *args, **kwargs)            # not a trigger we can reason about: the view answers (e.g. 400)

        cache, ttl = _cache(), int(_cfg("ACTIVE_TTL", 900))
        retry = int(_cfg("RETRY_AFTER", 15))
        inc_key = _incident_key(number)
        deadline = _monotonic() + float(_cfg("WAIT_SECONDS", 10))

        # ---- 1. incident lock: attach to the run that already exists, or wait for the one being started -------------
        while True:
            running = cache.get(inc_key)
            if isinstance(running, dict) and running.get("payload") is not None and _verified_running(running):
                log.info("triage_guard: duplicate trigger for %s attached to the running triage", number)
                return _same_run(running["payload"], "running")
            if isinstance(running, dict) and running.get("payload") is not None:
                _release(number, finished_ok=True)           # checker says it already finished: lock was stale
                continue
            if running is None:
                done = cache.get(_done_key(number))
                if done is not None:
                    return _same_run(done, "finished")        # REUSE_FINISHED_SECONDS: a very recent completed run
                break                                        # nobody is running it: go and claim it below
            # state == "starting": another request is being forwarded right now -> wait for it, do not forward again
            if _monotonic() >= deadline:
                return _busy(f"A triage for {number} is already being started. Please wait a moment.", 2, "already_starting")
            _sleep(0.1)

        # ---- 2. per-client burst limit --------------------------------------------------------------------------
        wait = _rate_limited(_client_of(request))
        if wait is not None:
            log.warning("triage_guard: rate limit hit by %s", _client_of(request))
            return _busy("Too many triage requests in a short time.", wait, "rate_limit")

        # ---- 3. claim the incident atomically (lost the race -> become a duplicate and retry the loop) -------------
        if not cache.add(inc_key, {"state": "starting", "ts": _now()}, ttl):
            return wrapper(request, *args, **kwargs)
        slot = _acquire_slot(number)
        if slot is None:
            cache.delete(inc_key)
            log.warning("triage_guard: no free slot for %s", number)
            return _busy("The triage service is busy with other incidents. Please try again shortly.", retry, "max_active")

        # ---- 4. forward. Anything but a clean 2xx frees everything again so the user can retry immediately ---------
        try:
            response = view(request, *args, **kwargs)
        except BaseException:
            _release_claim(number, slot)
            raise
        if not (200 <= response.status_code < 300):
            _release_claim(number, slot)
            return response

        payload, session_id = None, None
        try:
            payload = json.loads(response.content.decode(response.charset or "utf-8"))
            if isinstance(payload, dict):
                session_id = payload.get("session_id") or payload.get("id_trg_ssin") or payload.get("id")
        except Exception:
            payload = None
        if _is_finished(payload):                            # the start call already did all the work: nothing left to protect
            _release_claim(number, slot)
            if _is_success(payload):
                _remember_finished(number, payload)
            return response
        cache.set(inc_key, {"state": "running", "slot": slot, "session_id": session_id, "payload": payload, "ts": _now()}, ttl)
        if session_id:
            cache.set(_session_key(session_id), number.upper(), ttl)
        return response

    return wrapper
